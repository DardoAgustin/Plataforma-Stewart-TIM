#include <Arduino.h>
#include <ESP32Servo.h>
#include <Wire.h>
#include <MPU6050_light.h>

#include <hexapod_Arduino.h>
#include <Hexapod_GPIO.h>
#include <Hexapod_Kinematics.h>
#include <Hexapod_Servo.h>
#include <Hexapod_Config_1.h>

angle_t servo_angles[NB_SERVOS];
Hexapod_Servo hx_servo;
Hexapod_GPIO hx_gpio;

MPU6050 mpu(Wire);
bool mpuOk = false;

float Z_actual = 0.0, R_actual = 0.0, P_actual = 0.0;
bool modoMPU = false;
int kp = 1.1;

// Variables para suavizar el movimiento del MPU (Filtro Complementario)
float roll_suave = 0.0;
float pitch_suave = 0.0;
const float ALPHA = 0.20; // Cuanto más cerca de 1, más suave (pero más lento en responder)

/**
 * Mueve la plataforma a coordenadas (X, Y, Z, roll, pitch, yaw).
 */
void moverPlataforma(float X, float Y, float Z, float roll, float pitch, float yaw)
{
    // Límites de seguridad
    Z = constrain(Z, -10.0, 20.0);
    roll = constrain(roll, -15.0, 15.0);
    pitch = constrain(pitch, -15.0, 15.0);

    platform_t coord;
    coord.hx_x = X;
    coord.hx_y = Y;
    coord.hx_z = Z;
    coord.hx_a = roll  * DEG_TO_RAD;
    coord.hx_b = pitch * DEG_TO_RAD;
    coord.hx_c = yaw   * DEG_TO_RAD;

    int8_t movOK = hx_servo.calcServoAngles(coord, servo_angles);
    
    // Solo imprimimos si no estamos en modo MPU para no saturar el puerto serie
    if (!modoMPU) {
        Serial.print(">>> Pedido -> Z:"); Serial.print(Z);
        Serial.print(" | Roll:"); Serial.print(roll);
        Serial.print(" | Pitch:"); Serial.print(pitch);
        Serial.print(" | Estado:"); Serial.println(movOK); 
        
        if (movOK == 0) {
            Serial.print("Angulos Servos: ");
            for (int i = 0; i < NB_SERVOS; i++) { 
                Serial.print(servo_angles[i].deg, 1); 
                Serial.print("\t"); 
            }
            Serial.println("\n");
        } else {
            Serial.println("[ERROR] La cinemática no pudo calcular esta posición.");
        }
    }
    
    if (movOK == 0) {
        Z_actual = Z;
        R_actual = roll;
        P_actual = pitch;
    }
    
    hx_servo.updateServos(movOK);
}

void setup()
{
    Serial.begin(115200);
    delay(1000); 

    Serial.println("\n=== PLATAFORMA ESTABILIZADORA ===");
    Serial.println("Escribi 'M' para activar/desactivar el MODO MPU.");
    Serial.println("Escribi 'H' para volver a HOME.");
    Serial.println("Escribi comandos manuales: Z 10, R 5, P -5");
    Serial.println("================================");

    hx_gpio.setupGPIO();
    hx_servo.setupServo();
    delay(1000);

    // Inicializar el MPU6050 en los pines I2C por defecto (21 y 22)
    Serial.println("Inicializando MPU6050...");
    Wire.begin(); 
    byte status = mpu.begin(0, 0); // 0=max rango, 0=max rango
    if (status == 0) {
        mpuOk = true; 
        mpu.setFilterGyroCoef(0.90f); // Filtro interno del giroscopio
        Serial.println("Calibrando MPU... NO MOVER LA BASE.");
        delay(2000);
        mpu.calcOffsets(true, true); // Calibra acelerómetro y giroscopio
        Serial.println("MPU Calibrado.");
    } else {
        Serial.println("ERROR: MPU6050 no detectado. Revisa las conexiones I2C.");
    }

    // Arrancamos en HOME
    moverPlataforma(0, 0, 0, 0, 0, 0);
}

void loop()
{
    // Actualizar los datos del MPU en cada iteración
    if (mpuOk) mpu.update();

    // --- LECTURA DEL PUERTO SERIE ---
    if (Serial.available() > 0) 
    {
        String comando = Serial.readStringUntil('\n');
        comando.trim(); 
        if (comando.length() == 0) return;

        char eje = toupper(comando.charAt(0)); 

        if (eje == 'M') {
            modoMPU = !modoMPU;
            if (modoMPU) {
                Serial.println("\n[MODO MPU ACTIVADO]");
                // Inicializamos las variables suaves con la lectura actual
                roll_suave = mpu.getAngleX();
                pitch_suave = mpu.getAngleY();
            } else {
                Serial.println("\n[MODO MPU DESACTIVADO]");
            }
            return;
        }

        if (eje == 'H') {
            Serial.println("\n[COMANDO] Volviendo a HOME...");
            modoMPU = false;
            moverPlataforma(0, 0, 0, 0, 0, 0);
            return;
        }

        if (!modoMPU && (eje == 'Z' || eje == 'R' || eje == 'P')) {
            int espacioIndex = comando.indexOf(' ');
            if (espacioIndex > 0) {
                float valor = comando.substring(espacioIndex + 1).toFloat();
                Serial.println();
                if (eje == 'Z') moverPlataforma(0, 0, valor, R_actual, P_actual, 0);
                else if (eje == 'R') moverPlataforma(0, 0, Z_actual, valor, P_actual, 0);
                else if (eje == 'P') moverPlataforma(0, 0, Z_actual, R_actual, valor, 0);
            }
        }
    }

    // --- LÓGICA DE ESTABILIZACIÓN CON MPU ---
    if (modoMPU && mpuOk) {
        // Leemos los ángulos crudos del sensor
        float raw_roll = mpu.getAngleX();
        float raw_pitch = mpu.getAngleY();

        // 1. Aplicamos el filtro para suavizar las lecturas
        roll_suave = (ALPHA * roll_suave) + ((1.0 - ALPHA) * raw_roll);
        pitch_suave = (ALPHA * pitch_suave) + ((1.0 - ALPHA) * raw_pitch);

        // 2. Mapeo de ejes y compensación (según lo que me indicaste)
        // X_cin = -Y_mpu (Pitch de la base)
        // Y_cin = X_mpu (Roll de la base)
        float roll_compensacion = pitch_suave*kp; 
        float pitch_compensacion = -roll_suave*kp;

        // 3. Movemos la plataforma (manteniendo la altura actual)
        moverPlataforma(0, 0, Z_actual, roll_compensacion, pitch_compensacion, 0);
        
        // Opcional: Imprimir para debug cada 500ms
        static unsigned long lastPrint = 0;
        if (millis() - lastPrint > 500) {
            lastPrint = millis();
            Serial.print("MPU -> Roll:"); Serial.print(raw_roll);
            Serial.print(" Pitch:"); Serial.print(raw_pitch);
            Serial.print(" | Compensa -> R:"); Serial.print(roll_compensacion);
            Serial.print(" P:"); Serial.println(pitch_compensacion);
        }
    }
}
